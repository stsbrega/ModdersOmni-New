```python
# ---------------------------------------------------------------------------
# CPU family inference — when cpu_cores is None (common when the spec parser
# could not recover a core count from pasted text), use the model name to
# estimate cores. This prevents modern CPUs from scoring 0 just because the
# user did not paste a line like "CPU Cores: 6".
# ---------------------------------------------------------------------------
_CPU_FAMILY_CORES: list[tuple[re.Pattern, int]] = [
    # --- AMD Zen 4 / Zen 5 (7000/9000 series) ---
    (re.compile(r"Ryzen\s*5\s*[79]\d{3}", re.I), 6),   # Ryzen 5 7600/9600 → 6c
    (re.compile(r"Ryzen\s*7\s*[79]\d{3}", re.I), 8),   # Ryzen 7 7700/9700 → 8c
    (re.compile(r"Ryzen\s*9\s*[79]9\d{2}", re.I), 16), # Ryzen 9 7950/9950 → 16c
    (re.compile(r"Ryzen\s*9\s*[79]9\d{2}X3D", re.I), 16),
    (re.compile(r"Ryzen\s*9\s*[79]900X?3?D?", re.I), 12),  # Ryzen 9 7900/9900 → 12c
    # --- Intel Raptor Lake / Refresh (13th/14th gen Core i) ---
    (re.compile(r"Core\s*i5-1[34]6\d{2}", re.I), 14),  # i5-13600/14600 → 6P+8E
    (re.compile(r"Core\s*i7-1[34]7\d{2}", re.I), 20),  # i7-13700/14700 → 8P+12E (14th gen) / 8P+8E (13th gen). Use 20.
    (re.compile(r"Core\s*i9-1[34]9\d{2}", re.I), 24),  # i9-13900/14900 → 8P+16E
    # --- Intel Arrow Lake (Core Ultra 2xxK) ---
    (re.compile(r"Ultra\s*5\s*2[45]5K?", re.I), 14),   # 245K → 6P+8E
    (re.compile(r"Ultra\s*7\s*265K?", re.I), 20),      # 265K → 8P+12E
    (re.compile(r"Ultra\s*9\s*285K?", re.I), 24),      # 285K → 8P+16E
]


def _infer_cores_from_model(cpu: str | None) -> int | None:
    """When cpu_cores is not supplied, infer from the model name."""
    if not cpu:
        return None
    for pattern, cores in _CPU_FAMILY_CORES:
        if pattern.search(cpu):
            return cores
    return None
```

- [ ] **Step 2: Rewrite `_score_cpu` to use inference and an expanded high-perf list**

Replace `_score_cpu` (currently at `tier_classifier.py:65-107`) with:

```python
def _score_cpu(cpu: str | None, cores: int | None, speed_ghz: float | None) -> int:
    """Score CPU based on cores, clock speed, and known high-perf models (0-25).

    If `cores` is None but `cpu` matches a known family, infer cores from the
    model name so modern CPUs do not score 0 purely because the spec parse
    missed the core-count line.
    """
    score = 0

    # Resolve effective core count — supplied > inferred > none
    effective_cores = cores if cores is not None else _infer_cores_from_model(cpu)

    # Core count scoring (0-18)
    if effective_cores:
        if effective_cores >= 16:
            score = 18
        elif effective_cores >= 12:
            score = 15
        elif effective_cores >= 8:
            score = 12
        elif effective_cores >= 6:
            score = 8
        elif effective_cores >= 4:
            score = 5
        else:
            score = 2

    # Clock speed bonus (0-4)
    if speed_ghz:
        if speed_ghz >= 5.0:
            score += 4
        elif speed_ghz >= 4.5:
            score += 3
        elif speed_ghz >= 4.0:
            score += 2
        elif speed_ghz >= 3.5:
            score += 1

    # Known high-performance model bonus (+3)
    if cpu:
        high_perf = [
            # AMD Zen 4 X3D
            "7800X3D", "7950X3D",
            # AMD Zen 4 flagships
            "7950X", "7900X", "7900",
            # AMD Zen 5
            "9800X3D", "9950X", "9900X", "9700X",
            # Intel 13th/14th gen flagships
            "14900K", "13900K", "12900K", "14700K", "13700K",
            # Intel Arrow Lake
            "Ultra 9 285", "Ultra 7 265", "Ultra 9",
        ]
        for model in high_perf:
            if model.lower() in cpu.lower():
                score += 3
                break

    return min(score, 25)
```
