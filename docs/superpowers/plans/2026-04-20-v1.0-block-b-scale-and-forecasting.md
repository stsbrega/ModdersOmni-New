# v1.0 Block B — Scale Architecture, Disk-Space Forecasting, Alembic Adoption Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Deliver the scale feature (BYOK-gated Large/Massive/Ultimate presets up to 2000 mods via sub-pass splitting), post-generation disk-space forecasting, dashboard in-progress indicator, email on completion, and adopt Alembic for schema migrations.

**Architecture:** Extend the phased generation pipeline to run N sub-passes per phase when `effective_cap > 25`, where each sub-pass starts with fresh LLM context plus a compact "already added" summary. Capture `sizeInBytes` from Nexus on every `add_to_modlist` and aggregate through a new `/size-summary` endpoint. Dashboard polls a new `/status` endpoint every 15s to show in-progress badges. A pipeline terminal-state hook dispatches a completion email.

**Tech Stack:** Alembic for migrations, FastAPI + SQLAlchemy async, asyncio semaphores for Nexus throttling, existing SMTP infra reused for email, Angular 19 signals for polling.

**Spec reference:** `docs/superpowers/specs/2026-04-20-v1.0-release-design.md` § Block B + § Migration strategy.

**Prerequisite:** Block A should be merged first (storage_drives UI is needed for the size comparison bar).

```python
"""Modlist size preset — controls per-phase effective cap via multiplier."""

from enum import Enum


class SizePreset(str, Enum):
    SMALL = "small"
    MEDIUM = "medium"
    LARGE = "large"
    MASSIVE = "massive"
    ULTIMATE = "ultimate"

    @property
    def requires_byok(self) -> bool:
        """Presets above Medium require a user-supplied LLM API key."""
        return self in (SizePreset.LARGE, SizePreset.MASSIVE, SizePreset.ULTIMATE)


_MULTIPLIERS: dict[SizePreset, float] = {
    SizePreset.SMALL:    0.5,
    SizePreset.MEDIUM:   1.0,
    SizePreset.LARGE:    3.3,
    SizePreset.MASSIVE:  6.7,
    SizePreset.ULTIMATE: 13.3,
}

_TARGETS: dict[SizePreset, int] = {
    SizePreset.SMALL:    75,
    SizePreset.MEDIUM:   150,
    SizePreset.LARGE:    500,
    SizePreset.MASSIVE:  1000,
    SizePreset.ULTIMATE: 2000,
}


def preset_multiplier(preset: SizePreset) -> float:
    return _MULTIPLIERS[preset]


def preset_target(preset: SizePreset) -> int:
    """Approximate total mod count target for this preset (empirical calibration)."""
    return _TARGETS[preset]
```

```python
# backend/app/services/generation/sub_pass.py
"""Sub-pass planning for large phases.

When a phase's effective cap exceeds PER_SUB_PASS_MAX, split into N sub-passes
so each LLM turn starts with fresh context.
"""

import math
from dataclasses import dataclass

PER_SUB_PASS_MAX = 25  # mods per sub-pass; past this, context bloat hurts quality


@dataclass(frozen=True)
class SubPassPlan:
    effective_cap: int
    sub_pass_count: int
    per_sub_pass_cap: int  # the cap for each sub-pass (may differ slightly for the last)


def plan_sub_passes(
    *,
    phase_max_mods: int,
    phase_scale_weight: float,
    preset_multiplier: float,
) -> SubPassPlan:
    """Compute how many sub-passes a phase needs and the per-pass cap.

    Formula:
      effective_cap = round(phase_max_mods × phase_scale_weight × preset_multiplier)
      sub_pass_count = ceil(effective_cap / PER_SUB_PASS_MAX)
      per_sub_pass_cap = ceil(effective_cap / sub_pass_count)
    """
    effective_cap = round(phase_max_mods * phase_scale_weight * preset_multiplier)
    if effective_cap <= 0:
        return SubPassPlan(effective_cap=0, sub_pass_count=0, per_sub_pass_cap=0)
    sub_pass_count = max(1, math.ceil(effective_cap / PER_SUB_PASS_MAX))
    per_sub_pass_cap = math.ceil(effective_cap / sub_pass_count)
    return SubPassPlan(
        effective_cap=effective_cap,
        sub_pass_count=sub_pass_count,
        per_sub_pass_cap=per_sub_pass_cap,
    )
```

- [ ] **Step 2: Implement the endpoint**

In `backend/app/api/modlist.py`:

```python
from app.schemas.modlist import SizeSummary

INSTALLED_MULTIPLIER = 2.5
TIGHT_THRESHOLD = 0.80  # installed / free > 80% → tight
INSUFFICIENT_THRESHOLD = 1.00  # > 100% → insufficient


@router.get("/{modlist_id}/size-summary", response_model=SizeSummary)
async def get_size_summary(
    modlist_id: uuid.UUID,
    current_user: User = Depends(get_current_user),
    db: AsyncSession = Depends(get_db),
) -> SizeSummary:
    # Load modlist + ownership check
    result = await db.execute(
        select(Modlist).where(
            Modlist.id == modlist_id,
            Modlist.user_id == current_user.id,
        ).options(selectinload(Modlist.entries))
    )
    modlist = result.scalar_one_or_none()
    if modlist is None:
        raise HTTPException(status_code=404, detail="Modlist not found")

    sized = [e for e in modlist.entries if e.file_size_bytes is not None]
    unsized = [e for e in modlist.entries if e.file_size_bytes is None]
    download_bytes = sum(e.file_size_bytes for e in sized)
    installed_estimate_bytes = int(download_bytes * INSTALLED_MULTIPLIER)

    # Parse largest free drive from user settings
    largest_free = _parse_largest_free_drive(current_user.settings.storage_drives)

    # Classify fit
    if largest_free is None:
        fit_status = "unknown"
    elif installed_estimate_bytes > largest_free * INSUFFICIENT_THRESHOLD:
        fit_status = "insufficient"
    elif installed_estimate_bytes > largest_free * TIGHT_THRESHOLD:
        fit_status = "tight"
    else:
        fit_status = "comfortable"

    return SizeSummary(
        download_bytes=download_bytes,
        installed_estimate_bytes=installed_estimate_bytes,
        sized_count=len(sized),
        unsized_count=len(unsized),
        total_count=len(modlist.entries),
        largest_drive_free_bytes=largest_free,
        fit_status=fit_status,
    )


def _parse_largest_free_drive(storage_drives: str | None) -> int | None:
    """Parse 'C: 180GB free / 931GB, D: 450GB free / 1863GB' → bytes of largest free."""
    import re
    if not storage_drives:
        return None
    matches = re.findall(r"(\d+)\s*GB\s*free", storage_drives, re.IGNORECASE)
    if not matches:
        return None
    largest_gb = max(int(m) for m in matches)
    return largest_gb * 1024**3
```

```python
    """Dispatch a generation-complete email if the user is opted in."""
    prefs = (user.settings.notification_prefs or {}) if user.settings else {}
    if not prefs.get("email_on_generation_complete", True):
        logger.info("User %s opted out of completion emails — skipping.", user.id)

    game_name = modlist.game.name_display if getattr(modlist, "game", None) else "modlist"
    mod_count = len(modlist.entries) if modlist.entries else 0

    if outcome == "success":
        subject = f"Your {game_name} modlist is ready"

            f"— ModdersOmni"
        )
        html = (
            f"<p>Hey {user.display_name or 'there'},</p>"
            f"<p>Your <strong>{game_name}</strong> modlist just finished generating — "
            f"<strong>{mod_count} mods, {total_gb:.1f} GB download</strong>.</p>"
            f"<p><a href=\"https://moddersomni.com/modlist/{modlist.id}\">Open the modlist</a></p>"
            f"<p>— ModdersOmni</p>"
        )
    else:
        subject = f"Your {game_name} modlist generation ran into a problem"
        detail = f"\n\nDetails: {error_message}" if error_message else ""
        text = (
            f"Hey {user.display_name or 'there'},\n\n"
            f"Your {game_name} modlist generation didn't complete successfully."
            f"{detail}\n\n"
            f"You can retry from https://moddersomni.com/dashboard\n\n"
            f"— ModdersOmni"
        )
        html = (
            f"<p>Hey {user.display_name or 'there'},</p>"
            f"<p>Your <strong>{game_name}</strong> modlist generation didn't complete successfully."
            f"{('<br><small>' + error_message + '</small>') if error_message else ''}</p>"
            f"<p><a href=\"https://moddersomni.com/dashboard\">Retry from your dashboard</a></p>"
            f"<p>— ModdersOmni</p>"
        )

    await _smtp_send(to=user.email, subject=subject, html=html, text=text)
```
