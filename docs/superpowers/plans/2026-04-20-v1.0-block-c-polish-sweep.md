```ts
import { ChangeDetectionStrategy, Component, inject, signal, effect } from '@angular/core';
import { CommonModule } from '@angular/common';
import { AuthService } from '../../../core/services/auth.service';

const DISMISS_KEY = 'moddersomni:email-banner-dismissed';

@Component({
  selector: 'app-email-verification-banner',
  standalone: true,
  imports: [CommonModule],
  changeDetection: ChangeDetectionStrategy.OnPush,
  template: `
    @if (visible()) {
      <div class="banner" role="status">
        <span>Verify your email to secure your account.</span>
        <button class="resend" (click)="resend()" [disabled]="resending()">
          {{ resending() ? 'Sending…' : 'Resend verification link' }}
        </button>
        <button class="dismiss" (click)="dismiss()" aria-label="Dismiss">✕</button>
      </div>
    }
  `,
  styles: [`
    .banner { position: sticky; top: 0; z-index: 40; display: flex; align-items: center; gap: 1rem; justify-content: center; padding: 0.6rem 1rem; background: #c8831a; color: #fff; font-size: 0.9rem; }
    button { background: rgba(0,0,0,0.2); border: 1px solid rgba(255,255,255,0.4); color: #fff; padding: 0.3rem 0.7rem; border-radius: 0.3rem; cursor: pointer; font-size: 0.8rem; }
    button.dismiss { background: transparent; border: none; padding: 0.2rem 0.4rem; }
  `],
})
export class EmailVerificationBannerComponent {
  protected readonly auth = inject(AuthService);
  protected readonly dismissed = signal(localStorage.getItem(DISMISS_KEY) === 'true');
  protected readonly resending = signal(false);

  protected readonly visible = () => {
    const user = this.auth.currentUser();
    return !!user && !user.email_verified && !this.dismissed();
  };

  dismiss() {
    localStorage.setItem(DISMISS_KEY, 'true');
    this.dismissed.set(true);
  }

  async resend() {
    this.resending.set(true);
    try {
      await this.auth.resendVerificationEmail();
    } finally {
      this.resending.set(false);
    }
  }
}
```

## Task 11: Implement `DELETE /api/user/me`

**Files:** `backend/app/api/user.py`, cascade config in `backend/app/models/`

- [ ] **Step 1: Ensure FK cascades are configured**

Check `Modlist.user_id`:

```python
user_id: Mapped[uuid.UUID | None] = mapped_column(
    UUID(as_uuid=True),
    ForeignKey("users.id", ondelete="CASCADE"),  # was SET NULL per CLAUDE.md
    nullable=True,
)
```

Current codebase uses `ondelete="SET NULL"` on modlists (per the model file). Decision: for account deletion, we want the modlists **deleted** along with the user (user asked for "permanently deletes your account, all modlists, and all settings"). Either:

- Change `ondelete="SET NULL"` → `ondelete="CASCADE"` in the model (requires a new Alembic migration)
- OR keep SET NULL at the DB level and manually delete modlists in the endpoint

Pick **manual deletion in the endpoint** — it's explicit and lets other nullable user_id semantics stay intact (e.g., anonymous/legacy modlists). No schema change needed.

- [ ] **Step 2: Implement the endpoint**

In `backend/app/api/user.py`:

```python
from fastapi import APIRouter, Depends, Header, HTTPException, Response, status
from sqlalchemy import delete

from app.models.user import User
from app.models.user_settings import UserSettings
from app.models.modlist import Modlist, ModlistEntry, ModlistKnowledgeFlag
from app.api.deps import get_current_user, get_db

router = APIRouter(prefix="/api/user", tags=["user"])


@router.delete("/me", status_code=status.HTTP_204_NO_CONTENT)
async def delete_my_account(
    x_confirm_delete: str = Header(..., alias="X-Confirm-Delete"),
    current_user: User = Depends(get_current_user),
    db = Depends(get_db),
):
    if x_confirm_delete != "DELETE":
        raise HTTPException(
            status_code=400,
            detail="Missing or invalid X-Confirm-Delete header.",
        )

    # Cascade: delete child entries first, then modlists, then settings, then user.
    # SQLAlchemy relationships do NOT auto-cascade on ORM delete unless configured.
    user_modlist_ids = (await db.execute(
        select(Modlist.id).where(Modlist.user_id == current_user.id)
    )).scalars().all()

    if user_modlist_ids:
        await db.execute(
            delete(ModlistEntry).where(ModlistEntry.modlist_id.in_(user_modlist_ids))
        )
        await db.execute(
            delete(ModlistKnowledgeFlag).where(ModlistKnowledgeFlag.modlist_id.in_(user_modlist_ids))
        )
        await db.execute(
            delete(Modlist).where(Modlist.user_id == current_user.id)
        )

    await db.execute(delete(UserSettings).where(UserSettings.user_id == current_user.id))
    await db.execute(delete(User).where(User.id == current_user.id))
    await db.commit()

    return Response(status_code=204)
```

Ensure `router` is imported in `main.py` and registered.

```ts
    <div class="dialog" role="dialog" aria-modal="true">
      <h2>Delete your account?</h2>
      <p>This permanently deletes your account, all modlists, and all settings. This cannot be undone.</p>
      <label>
        Type <strong>DELETE</strong> to confirm:
        <input type="text" [(ngModel)]="confirmText" />
      </label>
      <div class="actions">
        <button class="cancel" (click)="cancel.emit()" [disabled]="working()">Cancel</button>
        <button class="danger" [disabled]="confirmText !== 'DELETE' || working()" (click)="confirm()">
          {{ working() ? 'Deleting…' : 'Delete account' }}
        </button>
      </div>
      @if (error()) { <p class="err">{{ error() }}</p> }
    </div>

    .err { color: #e55; margin-top: 0.5rem; font-size: 0.9rem; }
  `],
})
export class DeleteAccountDialogComponent {
  @Output() cancel = new EventEmitter<void>();

  protected confirmText = '';
  protected readonly working = signal(false);
  protected readonly error = signal<string | null>(null);
  private readonly users = inject(UserService);
  private readonly auth = inject(AuthService);
  private readonly router = inject(Router);

  confirm() {
    this.working.set(true);
    this.error.set(null);
    this.users.deleteAccount().subscribe({
      next: () => {
        this.auth.signOut();
        this.router.navigate(['/']);
      },
      error: (err) => {
        this.error.set(err?.error?.detail ?? 'Failed to delete account.');
        this.working.set(false);
      },
    });
  }
}
```
